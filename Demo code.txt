# Handle NULL values
from pyspark.sql.functions import coalesce, lit

cleaned_df = (
    placements_df
    .withColumn(
        "company",
        coalesce(col("company"), lit("Not Applicable"))
    )
    .withColumn(
        "job_role",
        coalesce(col("job_role"), lit("Not Applicable"))
    )
)

display(cleaned_df)

11
#Convert the date column
from pyspark.sql.functions import to_date

cleaned_df = cleaned_df.withColumn(
    "offer_date",
    to_date(col("offer_date"), "yyyy-MM-dd")
)

cleaned_df.printSchema()

12
# Create a package category
from pyspark.sql.functions import when

enriched_df = cleaned_df.withColumn(
    "package_category",
    when(col("annual_package") >= 1500000, "Very High")
    .when(col("annual_package") >= 1000000, "High")
    .when(col("annual_package") >= 700000, "Medium")
    .when(col("annual_package") > 0, "Entry Level")
    .otherwise("No Offer")
)

display(enriched_df)

Batch-wise statistics
%sql

SELECT
    batch_year,
    COUNT(*) AS total_students,
    SUM(
        CASE
            WHEN placement_status = 'Selected' THEN 1
            ELSE 0
        END
    ) AS selected_students,
    ROUND(AVG(
        CASE
            WHEN annual_package > 0 THEN annual_package
        END
    ), 2) AS average_package
FROM workspace.placement_demo.placement_results
GROUP BY batch_year
ORDER BY batch_year;

16
Create a reusable SQL view
%sql

CREATE OR REPLACE VIEW
workspace.placement_demo.program_placement_summary
AS
SELECT
    program,
    batch_year,
    COUNT(*) AS total_students,
    SUM(
        CASE
            WHEN placement_status = 'Selected' THEN 1
            ELSE 0
        END
    ) AS selected_students,
    ROUND(AVG(
        CASE
            WHEN annual_package > 0 THEN annual_package
        END
    ), 2) AS average_package,
    MAX(annual_package) AS highest_package
FROM workspace.placement_demo.placement_results
GROUP BY program, batch_year;

%sql

SELECT *
FROM workspace.placement_demo.program_placement_summary;
